#!/usr/bin/env python3
"""#4320 chunk 7: one fresh isolated Codex session per language reviews the FINAL search blocks of the
EnviousMeet Live Preview places after the rename "Live words" -> "Live Preview" (accuracy,
naturalness, collisions). Adapted from additions/meetingTranscriptAutoDetect/review-driver.py.txt.
Usage: rename-review-driver.py <worktree> <outdir>"""
import concurrent.futures
import json
import os
import pathlib
import subprocess
import sys
import tempfile

wt = pathlib.Path(sys.argv[1])
out = pathlib.Path(sys.argv[2])
out.mkdir(parents=True, exist_ok=True)
add = wt / "scripts/settings-map/receipts/additions"

IDS = ["meetingLiveWords", "meetingLiveWords.apple", "meetingLiveWords.change",
       "meetingLiveWords.keepCurrent", "meetingLiveWords.universal"]
TRUTH = {
    "meetingLiveWords": (
        "The live engine card on the Models tab of the EnviousMeet (meeting recorder) page. "
        "English title \"Live Preview\" (German \"Live-Vorschau\"); English description \"The engine "
        "and language for words shown while a meeting records.\" It chooses which engine (Apple or "
        "Universal) shows a rough draft of the words on screen while a meeting records, and holds "
        "the row for that live text's language. The feature was called \"Live words\" in English "
        "and \"Live-Transkript\" in German before; it is now named Live Preview, the same name the "
        "app uses for dictation's on-screen words, but it is a SEPARATE setting from dictation's "
        "Live Preview. It does NOT change the full transcript written after the meeting."),
    "meetingLiveWords.apple": (
        "The Apple choice inside the meeting Live Preview engine card (title \"Apple\"): use Apple's "
        "built-in speech engine for the meeting's Live Preview."),
    "meetingLiveWords.universal": (
        "The Universal choice inside the meeting Live Preview engine card (title \"Universal\"): use "
        "the downloadable Universal engine for the meeting's Live Preview."),
    "meetingLiveWords.change": (
        "The Change button on the meeting Live Preview engine card (English title \"Change\"). It opens "
        "the engine choices (Apple, Universal) for EnviousMeet's Live Preview."),
    "meetingLiveWords.keepCurrent": (
        "The button that closes the meeting Live Preview engine choices without changing them "
        "(English title \"Keep current Live Preview engine\")."),
}
SIBLINGS = ["livePreview", "section.livePreview", "dictation.tab.livePreview",
            "meetingLiveWordsLanguage", "meetingTranscriptLanguage", "meetingTranscriptEngine"]

vocab = json.load(
    open(wt / "Sources/EnviousWisprAppKit/Resources/SettingsSearchVocabulary.json", encoding="utf-8"))
by_id = {e["id"]: {b["language"]: b for b in e["blocks"]} for e in vocab["entries"]}
final = {i: json.load(open(add / f"{i}.json", encoding="utf-8"))["blocks"] for i in IDS}
langs = json.loads(sys.argv[3]) if len(sys.argv) > 3 else None
# Affected = the blocks that differ from chunk 6's commit (554f1470), per language.
base = {i: json.loads(subprocess.check_output(
    ["git", "-C", str(wt), "show", f"554f1470:scripts/settings-map/receipts/additions/{i}.json"]))["blocks"]
    for i in IDS}
affected = {}
for i in IDS:
    for lang, block in final[i].items():
        if base[i].get(lang) != block:
            affected.setdefault(lang, []).append(i)

BRIEF = """You are reviewing the search vocabulary blocks for a few places in the Settings search box of EnviousWispr, a macOS dictation app (press a key and speak, it types your words into any app). It also has EnviousMeet, a meeting recorder on its own page. The language you review is "{lang}".

Use ONLY this brief and the data below. Do not open, search or read any file, folder, repository or website. Do not run commands.

Context: EnviousMeet's on-screen meeting text was renamed from "Live words" to "Live Preview" (German: from "Live-Transkript" to "Live-Vorschau"). Dictation already has its own, separate Live Preview setting. The blocks below are the final blocks after a first rename pass; review them as they are now.

The places (English titles and descriptions are the source of truth):
{truth}

Review each block for language "{lang}" below:
1. Accuracy: remove or rewrite anything the place does not do.
2. Collisions: a block must not fit a sibling place equally well. Each block must make clear it is about meetings/EnviousMeet, not dictation's Live Preview. The siblings' current blocks in this language are listed below.
3. Naturalness: wording a native speaker of "{lang}" would really type or see in an app's settings. For a title: how an app in that language would label this place, naming the feature the way this app names Live Preview in that language. Keep product names (EnviousMeet, EnviousWispr, Apple, Universal, macOS, Mac) unchanged. Lowercase except nouns the language capitalises. "words" 0 to 6 short terms; "phrases" 1 to 3 everyday searches, at least one. For en and de there is no title (the interface supplies it); for de use du-form. An old name kept as a search word on the card (meetingLiveWords) is intentional, so people who remember it still find the card; do not add the old name to the other places.

Blocks to review (id -> block):
{blocks}

Sibling blocks in "{lang}" (id -> block):
{siblings}

Output: one JSON object only, no commentary, no code fence, mapping EVERY id above to its corrected block in exactly the shape given (keep "language"; omit "title" for en and de). If a block is already right, return it unchanged. After the JSON add nothing.
"""


def run(lang: str) -> tuple[str, int]:
    ids = affected[lang]
    prompt = BRIEF.format(
        lang=lang,
        truth="\n".join(f"- {i}: {TRUTH[i]}" for i in ids),
        blocks=json.dumps({i: final[i][lang] for i in ids}, ensure_ascii=False, indent=1),
        siblings=json.dumps(
            {i: by_id[i][lang] for i in SIBLINGS if lang in by_id.get(i, {})},
            ensure_ascii=False, indent=1),
    )
    (out / f"prompt-{lang}.txt").write_text(prompt, encoding="utf-8")
    iso = pathlib.Path(tempfile.mkdtemp(prefix="ew-vocab-review."))
    (iso / "work").mkdir()
    (iso / "codex-home").mkdir()
    os.symlink(os.path.expanduser("~/.codex/auth.json"), iso / "codex-home" / "auth.json")
    answer = iso / "out.txt"
    env = dict(os.environ, CODEX_HOME=str(iso / "codex-home"))
    proc = subprocess.run(
        [os.path.expanduser("~/.claude/bin/codex-run"), str(answer), "-c", "tools.web_search=false",
         "exec", "--sandbox", "read-only", "--skip-git-repo-check", "--ignore-rules", "--ephemeral",
         "-C", str(iso / "work")],
        input=prompt, text=True, env=env, cwd=str(iso / "work"), capture_output=True)
    last = pathlib.Path(str(answer) + ".last")
    if proc.returncode == 0 and last.exists():
        (out / f"review-{lang}.txt").write_text(last.read_text(encoding="utf-8"), encoding="utf-8")
    else:
        (out / f"review-{lang}.FAILED").write_text(
            f"exit {proc.returncode}\n{proc.stderr[-2000:]}", encoding="utf-8")
    return lang, proc.returncode


all_langs = sorted(affected)
todo = langs or all_langs
with concurrent.futures.ThreadPoolExecutor(max_workers=6) as pool:
    for lang, rc in pool.map(run, todo):
        print(f"{lang}: exit {rc}", flush=True)
print("DONE", flush=True)
