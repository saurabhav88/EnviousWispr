#!/usr/bin/env python3
"""#3973 PR B (driver copied from #4058's): one fresh isolated Codex session per language and place reviews
the drafted Models tab block
(accuracy, naturalness, collisions with sibling places). Mirrors scripts/settings-map/draft-vocabulary.sh
isolation. Usage: review4058.py <worktree> <drafts-dir> <outdir>"""
import concurrent.futures
import glob
import json
import os
import pathlib
import subprocess
import sys
import tempfile

wt = pathlib.Path(sys.argv[1])
drafts = pathlib.Path(sys.argv[2])
out = pathlib.Path(sys.argv[3])
vocab = json.load(open(wt / "Sources/EnviousWisprAppKit/Resources/SettingsSearchVocabulary.json", encoding="utf-8"))
by_id = {e["id"]: {b["language"]: b for b in e["blocks"]} for e in vocab["entries"]}

PAGE = ("EnviousMeet is the app's meeting recorder: it records a call or meeting on the Mac (your microphone and, "
        "if chosen, the sound the Mac plays) and makes a transcript with who said what, plus an optional summary. "
        "Its settings page has four tabs: General, Models, Call Apps, Permissions.")
PLACES = {
    "meetings.tab.models": dict(
        en_title="Models", de_title="Modelle", desc="(none)",
        does=("a TAB of the EnviousMeet page that holds every model choice for meetings: Live words (the engine and "
              "language for words shown while a meeting records: Apple speech or the Universal engine, with a "
              "Download button), Transcript language (the language of the full transcript written after a meeting, "
              "Auto or a chosen language), and Summaries (which AI Polish provider and model write the meeting "
              "summary; it follows the AI Polish settings). A Transcript engine choice (Parakeet, Whisper, Apple "
              "speech) will join it soon. It is NOT the dictation engine page, NOT AI Polish settings themselves and "
              "NOT where models for dictation are downloaded."),
        siblings=["meetings.tab.general", "meetingLiveWords", "meetingTranscriptLanguage", "transcriptionEngine",
                  "aiPolishProvider", "meetings.tab.callApps"]),
}

BRIEF = """You are reviewing a draft search vocabulary block for ONE place in the Settings search box of EnviousWispr, a macOS dictation app (press a key and speak, it types your words into any app). The language you review is "{lang}".

Use ONLY this brief and the data below. Do not open, search or read any file, folder, repository or website. Do not run commands.

{page}

The place (English title and description are the source of truth; the page is EnviousMeet):
- English title: {en_title}
- German title: {de_title}
- English description: {desc}
- What it really is: {does}

Review the draft block for language "{lang}" below:
1. Accuracy: remove or rewrite anything the place does not do or hold.
2. Collisions: the block must not fit the sibling places equally well. The siblings' current blocks in this language are listed below (a sibling may be missing when it is new).
3. Naturalness: wording a native speaker of "{lang}" would really type or see in an app's settings. For the title: how an app in that language would label this. Keep product names unchanged. Lowercase except nouns the language capitalises. "words" 0 to 6 short terms; "phrases" 1 to 3 everyday searches, at least one. For en and de there is no title (the interface supplies it); for de use du-form.

Draft block:
{draft}

Sibling blocks in "{lang}" (id -> block):
{siblings}

Output: one JSON object only, no commentary, no code fence, with the corrected block in exactly the draft's shape: {{"language": "{lang}", "title": "...", "words": [...], "phrases": [...]}} (omit "title" for en and de). If the draft is already right, return it unchanged. After the JSON add nothing.
"""


def draft_for(pid):
    if pid in by_id:  # a retained place: its shipped blocks are the draft under re-review
        return {"blocks": list(by_id[pid].values())}
    paths = sorted(glob.glob(str(drafts / f"{pid}-*" / "draft.json")))
    assert len(paths) == 1, (pid, paths)
    return json.load(open(paths[0], encoding="utf-8"))


def run(job):
    pid, lang, block = job
    p = PLACES[pid]
    prompt = BRIEF.format(
        lang=lang, page=PAGE, en_title=p["en_title"], de_title=p["de_title"], desc=p["desc"], does=p["does"],
        draft=json.dumps(block, ensure_ascii=False, indent=1),
        siblings=json.dumps({i: by_id[i][lang] for i in p["siblings"] if lang in by_id.get(i, {})},
                            ensure_ascii=False, indent=1))
    dest = out / pid
    dest.mkdir(parents=True, exist_ok=True)
    if (dest / f"review-{lang}.txt").exists():
        return pid, lang, 0
    iso = pathlib.Path(tempfile.mkdtemp(prefix="ew-vocab-review."))
    (iso / "work").mkdir()
    (iso / "codex-home").mkdir()
    os.symlink(os.path.expanduser("~/.codex/auth.json"), iso / "codex-home" / "auth.json")
    answer = iso / "out.txt"
    env = dict(os.environ, CODEX_HOME=str(iso / "codex-home"))
    proc = subprocess.run(
        [os.path.expanduser("~/.claude/bin/codex-run"), str(answer), "-c", "tools.web_search=false", "exec",
         "--sandbox", "read-only", "--skip-git-repo-check", "--ignore-rules", "--ephemeral", "-C", str(iso / "work")],
        input=prompt, text=True, env=env, cwd=str(iso / "work"), capture_output=True)
    last = pathlib.Path(str(answer) + ".last")
    if proc.returncode == 0 and last.exists():
        (dest / f"review-{lang}.txt").write_text(last.read_text(encoding="utf-8"), encoding="utf-8")
    else:
        (dest / f"review-{lang}.FAILED").write_text(f"exit {proc.returncode}\n{proc.stderr[-2000:]}", encoding="utf-8")
    return pid, lang, proc.returncode


jobs = []
for pid in PLACES:
    d = draft_for(pid)
    for b in d["blocks"]:
        jobs.append((pid, b["language"], b))
with concurrent.futures.ThreadPoolExecutor(max_workers=6) as pool:
    for pid, lang, rc in pool.map(run, jobs):
        print(f"{pid} {lang}: exit {rc}", flush=True)
print("DONE", flush=True)
